    const patiApp = new Vue({
        el: "#patiApp",
        data: {
            rgsInfo: {
                "dName": "",
                "dGender": "",
                "dMajor": "",
                "dDept": "",
                "dCost": "",
            },
            rgssInfo: [],
            addPopover: "none",
            delPopover: "none",
            updPopover: "none",
            delRID: "",
            updRID: "",
            filterCondition: {
                pName: "",
                dName: ""
            }
        },
        methods: {
            getAllRgssInfo(){
                let that = this;
                let condition = this.filterCondition;
                //1.将对象转为json字符串
                let conditionJson = JSON.stringify(condition);
                //2.json注入formData
                let formData = new FormData();
                formData.append("conditionJson",conditionJson);
                //3.自定义请求借口
                let option = {
                    url: "getAllRgssInfo",
                    method: "post",
                    data: formData
                }
                axios(option).then(function (res) {
                    if(res.data.status){
                        that.rgssInfo = res.data.list;
                        that.filterCondition = {};
                        console.log(that.rgssInfo);
                    }
                    else {
                        that.$message.error(res.data.msg);
                    }
                })
            },
            openAddPopover(){
                this.rgsInfo = {};
                this.addPopover = "flex";
            },
            noPopover(){
                this.addPopover = "none";
            },
            addRgsInfo(){
                let that = this;
                let rgsInfo = this.rgsInfo;
                //1.将对象转为json字符串
                let rgsInfoJson = JSON.stringify(rgsInfo);
                //2.json注入formData
                let formData = new FormData();
                formData.append("rgsInfoJson",rgsInfoJson);
                //3.自定义请求借口
                let option = {
                    url: "addRgsInfo",
                    method: "post",
                    data: formData
                }
                axios(option).then(function (res) {
                    if(res.data.status){
                        that.getAllRgssInfo();
                        that.noPopover();
                    }
                    else {
                        // that.$message.error(res.data.msg);
                        console.log(res.data.msg);
                    }
                })
            },
            openDelPopover(dID){
                this.delPopover='flex';
                this.delDID=dID;
            },
            delRgsInfo(){
                let dID = this.delDID;
                let that = this;
                axios.get("delRgsInfo?dID=" + dID).then(function (res) {
                    if(res.data.status){
                        that.delPopover = "none";
                        that.getAllRgssInfo();
                    }
                    else {
                        console.log(res.data.msg);
                    }
                })
            },
            openUpdPopover(pati){
                this.rgsInfo = pati;
                this.updPopover = "flex";
            },
            updRgsInfo(){
                let that = this;
                let rgsInfo = this.rgsInfo;
                //1.将对象转为json字符串
                let rgsInfoJson = JSON.stringify(rgsInfo);
                //2.json注入formData
                let formData = new FormData();
                formData.append("rgsInfoJson",rgsInfoJson);
                //3.自定义请求借口
                let option = {
                    url: "updRgsInfo",
                    method: "post",
                    data: formData
                }
                axios(option).then(function (res) {
                    if(res.data.status){
                        that.getAllRgssInfo();
                        that.updPopover = 'none';
                    }
                    else {
                        console.log(res.data.msg);
                    }
                })
            }
        },
        mounted(){
            this.getAllRgssInfo();
        }
    })
